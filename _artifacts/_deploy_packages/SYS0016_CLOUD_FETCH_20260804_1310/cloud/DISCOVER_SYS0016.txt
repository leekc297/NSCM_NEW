    SELECT DISTINCT REGEXP_SUBSTR(A.API_URL, '[^.]+', 1, 1)
    FROM   SCM_MGR.SCM_ADM_MNU_BTN_MPG_M B

    WHERE  B.MNU_ID = 'SYS0016' AND A.API_URL IS NOT NULL
    UNION

    WHERE  OWNER='SCM_MGR' AND OBJECT_NAME LIKE '%SYS0016%'

),
clo (OWNER, NAME, TYPE) AS (
    SELECT D.REFERENCED_OWNER, D.REFERENCED_NAME, D.REFERENCED_TYPE
    FROM   ALL_DEPENDENCIES D JOIN units U ON D.NAME = U.NAME
    WHERE  D.OWNER = 'SCM_MGR' AND D.REFERENCED_OWNER NOT IN ('SYS','PUBLIC','SYSTEM')
    UNION
    SELECT D.REFERENCED_OWNER, D.REFERENCED_NAME, D.REFERENCED_TYPE
    FROM   ALL_DEPENDENCIES D JOIN clo C
           ON D.OWNER = C.OWNER AND D.NAME = C.NAME
    WHERE  D.REFERENCED_OWNER NOT IN ('SYS','PUBLIC','SYSTEM')
)
