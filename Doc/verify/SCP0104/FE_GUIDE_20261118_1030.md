const namesOf = (rows: ComboRow[], idKey: string, nameKey: string): string[] => {
  const seen = new Set<string>();
  for (const row of rows) {

    const name = str(row[nameKey]);

### 변경 후 (예시)

```ts
import { sortMasterCodes } from './capaLineAssign.cascade';

/** 명칭 후보. AS-IS GetSortedCode 코드순(DMY_ 뒤) 표를 명칭으로 바꾼 순서(cs:1791~1806, ScpCapaBL.cs:81~136) */
const namesOf = (rows: ComboRow[], idKey: string, nameKey: string): string[] => {
  const byCode = new Map<string, string>();
  for (const row of rows) {
    const code = str(row[idKey]);
    const name = str(row[nameKey]);
    if (!code || code === EMPTY_CODE || !name || byCode.has(code)) continue;
    byCode.set(code, stripCodePrefix(code, name));
  }
  const ordered = sortMasterCodes([...byCode].map(([value, label]) => ({ value, label })));
  return [...new Set(ordered.map(option => option.label))];
};
```
